#Como usar este script:

#Salve o script: Salve o código acima em um arquivo, por exemplo, criar_estrutura.sh.

#Dê permissão de execução: chmod +x criar_estrutura.sh

#Execute o script: sudo ./criar_estrutura.sh (é necessário sudo para criar usuários, grupos e diretórios no nível do sistema).

#Configure as senhas: Após a execução, use o comando passwd nome_do_usuario para definir a senha de cada usuário.
